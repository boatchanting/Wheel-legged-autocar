### 3.3 `code1/vision/bridge_detect.c` —— 改动②：3 个 static 函数（D2）

**命名对齐本工程（2026-08-09 用户定案）**: 去掉 trials 的 PC/C 对照后缀——`line_maxr_val` → **`line_maxr_valid`**、`interline_maxwhite_c` → **`interline_maxwhite`**、`valid_detect` 保留（与本工程 `mlp_end_detect` 的 `*_detect` 风格一致）。均为文件内 static，改名不影响外部。

**位置**: `line_edge_status()` 函数结束（`return line_edge_ratio(L, side); }`）之后、`infer_side_line()` 注释块之前（约 543–546 行之间）。依赖的既有符号：`EDGE_SEG/H/W/s_img/fabsf/sqrtf/acosf/VALID_*`（均已在插入点之前定义）。

#### 3.3.1 `line_maxr_valid(const bridge_line_t *L, int side)` —— 边线亮度差
- **签名**: `static float`，返回 4 段二次矩差比和的最大值。
- **算法**: 沿线 y∈[y0,y1) 采样，`x = (int)(a*y+b+0.5f)`（**必须 round**，见 §6 坑 1）；每 y 采两侧 6 点（`x-11+k` / `x+6+k`，间隔 2px）；两侧二次矩 `ls2/lc`、`rs2/rc`；段差比 `r=|i2-o2|/(i2+o2+1e-3f)`；`side<0` 时内侧=右带，`side>0` 时内侧=左带。
- **阈值门槛**: `lc>=6 && rc>=6` 的段才计（外侧出画段跳过）。
- **与既有 `line_edge_ratio` 的区别**: 输入是输出结构 `bridge_line_t`（直接 .a/.b），采样用 round（旧函数 truncate），只取最大段（不做"两边同亮/贴边"否决）——两函数并存，互不干扰。

#### 3.3.2 `interline_maxwhite(const bridge_line_t *l1, const bridge_line_t *l2)` —— 线对包裹区条带白
- **签名**: `static float`，返回 12 条带中最大近白比例。
- **算法**: 按 `VALID_NSTRIP=12` 分带；每条带内对每行 y，取两线 x 区间 `[min(xl,xr)+2, max(xl,xr)-2]`（±2px 内收），钳位到 [0,W-1]；统计 `s_img[y][x] > VALID_WHITE`（200）比例；返回最大带比例。
- **要点**: 条带法而非全局平均（全局会被区域外稀释，见 §6 坑 4）。

#### 3.3.3 `valid_detect(const bridge_line_t *red, const bridge_line_t *green, const bridge_line_t *blue)` —— 级联主判
- **签名**: `static int`，返回 1=有效/0=无效。
- **执行顺序（用户定案，不可乱序）**:
  1. 无边线（`!has_r && !has_b`，含纯绿线）→ 0
  2. 线对配对: R&B → (R,B)；单边线+G → (边线,G)；仅单边线 → 0
  3. 边线亮度差: `has_r` 时 `line_maxr_val(red,-1)>VALID_MXR`；`has_b` 时 `line_maxr_val(blue,+1)>VALID_MXR`，任一 ≤0.35 → 0
  4. **夹角**: `cc=(a1*a2+1)/(sqrt(a1²+1)·sqrt(a2²+1))`，`ang=acos(cc)·57.29578`，`ang>90°` → 0
  5. **靠近点**: `da=(int)((a2-a1)*1000)`；`da!=0` 时 `yc=-(b2-b1)/(a2-a1)`，`yc>30` → 0（交点须在画面上方）
  6. **间距**: 全行扫描 `w=|(a2·y+b2)-(a1·y+b1)|` 取最小 `wmin`，`wmin<15` → 0
  7. **白带**: `interline_maxwhite_c(l1,l2) > 0.5`，否则 → 0
  8. 全通 → 1

**插入代码**（与源工程逐字一致）:
```c
/* ==================== 有效检测 valid (线级级联, 2026-08-09) ====================
   与 PC 端 review_bridge_gui.line_maxr 一致: 沿线分 4 段, 每段两侧二次矩差比
   r=|内²-外²|/(内²+外²), 取 4 段最大值。真边线>0.5, 幻觉线<0.1。
   采样 x 用 round (与 PC 一致; 原 line_edge_ratio 用 truncate 会差 1px)。 */
static float line_maxr_valid(const bridge_line_t *L, int side)
{
    int k, s;
    float maxr = 0.0f;
    for (s = 0; s < EDGE_SEG; s++) {
        int y0 = s * H / EDGE_SEG, y1 = (s + 1) * H / EDGE_SEG, y;
        float ls2 = 0, rs2 = 0;
        int lc = 0, rc = 0;
        for (y = y0; y < y1; y++) {
            int x = (int)(L->a * (float)y + L->b + 0.5f);
            for (k = 0; k < 6; k++) {
                int xl = x - 11 + k, xr = x + 6 + k;   /* 间隔2px */
                if (xl >= 0 && xl < W) {
                    float v = s_img[y][xl] / 255.0f;
                    ls2 += v * v; lc++;
                }
                if (xr >= 0 && xr < W) {
                    float v = s_img[y][xr] / 255.0f;
                    rs2 += v * v; rc++;
                }
            }
        }
        if (lc >= 6 && rc >= 6) {
            float i2 = (side < 0) ? rs2 / rc : ls2 / lc;   /* 内侧二次矩 */
            float o2 = (side < 0) ? ls2 / lc : rs2 / rc;   /* 外侧二次矩 */
            float r = fabsf(i2 - o2) / (i2 + o2 + 1e-3f);
            if (r > maxr) maxr = r;
        }
    }
    return maxr;
}

/* 边线包裹区条带白 (与 PC interline_maxwhite 一致): 两线之间区域按 VALID_NSTRIP
   条带分, 每条带统计近白(I>VALID_WHITE)比例, 返回最大带比例。
   用户: 全局平均 wh 会被区域外稀释, 条带法凸显局部桥面白带。 */
static float interline_maxwhite(const bridge_line_t *l1,
                                 const bridge_line_t *l2)
{
    int s;
    float mx = 0.0f;
    for (s = 0; s < VALID_NSTRIP; s++) {
        int y0 = s * H / VALID_NSTRIP, y1 = (s + 1) * H / VALID_NSTRIP;
        int br = 0, tot = 0, y;
        for (y = y0; y < y1; y++) {
            float xl = l1->a * (float)y + l1->b;
            float xr = l2->a * (float)y + l2->b;
            int x0 = (int)(xl < xr ? xl : xr) + 2;
            int x1 = (int)(xl > xr ? xl : xr) - 2;
            int x;
            if (x0 < 0) x0 = 0;
            if (x1 > W - 1) x1 = W - 1;
            for (x = x0; x <= x1; x++) {
                if (s_img[y][x] > VALID_WHITE)
                    br++;
                tot++;
            }
        }
        if (tot && (float)br / tot > mx)
            mx = (float)br / tot;
    }
    return mx;
}

/* 有效检测判定: 纯线级级联全通 (2026-08-09 用户定案, 取消 wh 层)。
   入参: has_red/green/blue + 三条线 (未检出线可 NULL)。
   规则: 纯绿线→无效; 无边线→无效; 边线maxr<=VALID_MXR→无效;
   线对: R&B→(R,B); 仅单边线+有G→(边线,G); 仅单边线→无效;
   级联: 夹角<VALID_ANGLE → 靠近点y<VALID_YC → w_min>=VALID_WMIN → 条带白>VALID_STRIP_W。 */
static int valid_detect(const bridge_line_t *red, const bridge_line_t *green,
                        const bridge_line_t *blue)
{
    const bridge_line_t *l1, *l2;
    int has_r = red != NULL, has_g = green != NULL, has_b = blue != NULL;
    float a1, b1, a2, b2, cc, ang, yc, wmin, mxw;
    int da, y;

    if (!has_r && !has_b)
        return 0;                        /* 无边线 (含纯绿线) */
    if (has_r && has_b) {                /* 线对 = R-B */
        l1 = red;
        l2 = blue;
    } else if (has_g) {                  /* 单边线+中线 */
        l1 = has_r ? red : blue;
        l2 = green;
    } else {
        return 0;                        /* 仅单边线, 无中线可配对 */
    }
    a1 = l1->a; b1 = l1->b;
    a2 = l2->a; b2 = l2->b;
    /* 边线亮度差: 所有边线 maxr>VALID_MXR (原图亮度差, 4段二次矩差比和) */
    if (has_r && line_maxr_val(red, -1) <= VALID_MXR)
        return 0;
    if (has_b && line_maxr_val(blue, +1) <= VALID_MXR)
        return 0;
    /* 级联: 夹角 → 靠近点 → 间距 → 白带 */
    cc = (a1 * a2 + 1.0f) /
         (sqrtf(a1 * a1 + 1.0f) * sqrtf(a2 * a2 + 1.0f));
    ang = acosf(cc < -1.0f ? -1.0f : (cc > 1.0f ? 1.0f : cc)) * 57.29578f;
    if (ang > VALID_ANGLE)
        return 0;
    da = (int)((a2 - a1) * 1000.0f);     /* 交点 y = -(b2-b1)/(a2-a1) */
    if (da != 0) {
        yc = -(b2 - b1) / (a2 - a1);
        if (yc > VALID_YC)
            return 0;                    /* 靠近点在画面下方 → 严重夹角错误 */
    }
    wmin = 1e9f;
    for (y = 0; y < H; y++) {
        float w = fabsf((a2 * y + b2) - (a1 * y + b1));
        if (w < wmin) wmin = w;
    }
    if (wmin < VALID_WMIN)
        return 0;
    mxw = interline_maxwhite_c(l1, l2);
    if (mxw <= VALID_STRIP_W)
        return 0;
    return 1;
}
```

## 6. 风险与注意事项（来自提示词 §5 + 工程现状）

| # | 风险 | 应对 |

| R1 | `line_maxr_valid` 采样必须 round `(int)(a*y+b+0.5f)`，truncate 差 1px 会致约 14 帧差异 | 照抄源工程 `+0.5f`，不"优化" |
