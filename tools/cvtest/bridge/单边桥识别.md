撰写py代码，使用这套类似的算法，读取png文件 ，进行单边桥识别
【场景】

在一片广阔而平坦的深红色地面上，横跨着一座长 300 厘米、宽 45 厘米、厚 2 厘米的白色矩形桥面。桥面水平居中放置，其表面清晰地标有三条垂直于行进方向的深灰色分段线，分别位于 -100 cm、0 cm 和 +100 cm 处，用以指示每 1 米的区段。

在桥面上，沿纵向（Z 轴方向）规则地布置了三个黑色楔形障碍物。这些障碍物呈“屋脊形”：底面为矩形（宽 27 cm，长 20 cm），顶部沿桥宽方向形成一条棱线，整体高度为 5 cm。它们并非对称排列，而是采用“左–右–左”的交错方式——第一个楔形紧贴桥面左侧边缘内侧，第二个紧贴右侧边缘内侧，第三个再次回到左侧。相邻两个楔形之间的空隙（即未被障碍物占据的桥面长度）固定为 20 厘米，符合 10~25 厘米的规范要求。

所有楔形均避开了桥头和桥尾区域，在距离两端至少 25 厘米的安全边距内布置，确保不会紧贴桥端。

/*-------------------------------------------------------------------------------------------------------------------
@brief   计算角度(最后的边线数组border)
@param   y          角点y坐标
         dy         y坐标差值
         dir        左右线

@return  角度
Sample   Cal_Angle_Degree_Border(50, 5, -1);
@note
-------------------------------------------------------------------------------------------------------------------*/
float Cal_Angle_Degree_Border(int y, int dy, int dir)
{
    float ans;

    if (dir==1)         //右线 R
        ans = (atanf((r_border[y-dy]-r_border[y])/dy) - atanf((r_border[y]-r_border[y+dy])/dy)) * DEG_TO_RAD;
    else if (dir==-1)   //左线 L
        ans = (atanf((l_border[y-dy]-l_border[y])/dy) - atanf((l_border[y]-l_border[y+dy])/dy)) * DEG_TO_RAD;

    return ans;
}

@brief   计算角度(原始二维边线数组)

         dir        左右线

@return  角度
Sample   Cal_Angle_Degree_Border(50, 10, -1);
@note
-------------------------------------------------------------------------------------------------------------------*/
float Cal_Angle_Degree_Points(int y, int dy, int dir)
{
    float ans;
    int index;

    if (dir==1)         //右线 R
    {
        ans = (atanf((points_r[y-dy][0]-points_r[y][0]) / (points_r[y-dy][1]-points_r[y][1])) - atanf((points_r[y][0]-points_r[y+dy][0]) / (points_r[y][1]-points_r[y+dy][1]))) * DEG_TO_RAD;

    }

    return ans;
}

/*-------------------------------------------------------------------------------------------------------------------
@brief   赛道连续性检测
@param   start      输入起始点纵坐标
         end        结束点纵坐标
         limit      断裂判断阈值
         dir        左右线

@return  跳变点的y坐标(尚未跳变的末端点)
Sample   Continuity_Change(110, 10, 5, -1)
@note
-------------------------------------------------------------------------------------------------------------------*/
int Continuity_Change(int start, int end, int limit, int dir)      //赛道连续性检测
{
    int i;
    int continuity_change_flag=0;

    if(start > image_start_row)//数组越界保护
        start = image_start_row;

    if(end<hightest)
        end=hightest;

    if (dir==1)         //右线 R
    {
        if (right_lost_time<0.9*image_h)  //丢线过多，无意义
            for(i=start;i>=end;i--)
            {
                if(r_border[i] >= right_stop_point[i]) continue;
                if(abs(r_border[i]-r_border[i-1])>=limit)
               {
                    continuity_change_flag=i;
                    break;
               }
            }
    }

    else if (dir==-1)   //左线 L
    {
        if (left_lost_time<0.9*image_h)  //丢线过多，无意义
            for(i=start;i>=end;i--)
            {
                if(l_border[i] <= left_stop_point[i]) continue;
                if(abs(l_border[i]-l_border[i-1])>=limit)
               {
                    continuity_change_flag=i;
                    break;
               }
            }
    }

    return continuity_change_flag;
}



/*-------------------------------------------------------------------------------------------------------------------
@brief   寻找单调性突变点

@return  0:无突变/丢线   非0:突变点

int Monotonicity_Change_Point(int start, int end, int dir)      //单调性突变点
{
    int index = 0;

    {

                }
            }
    }

    else if (dir==-1)   //左线
    {
        if (left_lost_time<0.9*image_h)  //丢线过多，无意义
            for (int i = start;i>end;i--)
            {
                if (l_border[i] > l_border[i+6] && l_border[i] > l_border[i-6] &&

                    index=i;    //角点对应的指针
                    break;
                }
            }
    }

    return index;
}



/*-------------------------------------------------------------------------------------------------------------------
@brief   斜率计算
@param   startX 起始点x坐标
         startY 起始点y坐标
         endX   结束点x坐标
         endY   结束点y坐标

@return  斜率    0为无穷
Sample   Cal_Slope_Border(20, 20, 60, 100);
@note
-------------------------------------------------------------------------------------------------------------------*/
float Cal_Slope_Border(int startX, int startY, int endX, int endY)
{
    float ans;

    if(startX == endX)  //斜率为无穷，用0代替
        ans = 0;
    else
        ans = (endY - startY) / (endX - startX);

    return ans;
}





/*-------------------------------------------------------------------------------------------------------------------
@brief   通过斜率，定点补线
@param   k      斜率
         startY 起始点y坐标
         endY   结束点y坐标
         dir    左右线

@return  null
Sample   K_Add_Boundry(1, 110, 80, -1);
@note    补的线直接修改边线，从下往上补线
-------------------------------------------------------------------------------------------------------------------*/
void K_Add_Boundry(float k, int startY, int endY, int dir)
{
    int i,startX;

    if(startY >= image_h) startY=image_h-1;
    if(endY < 0) endY=0;

    if (dir==1)     //右线 R
    {
        startX = r_border[startY];

        if (k==0)       //斜率为0则是斜率为无穷大的情况，所以垂直向上拉线
            for(i=startY+1;i>=endY;i--)
                        r_border[i]=startX; //(y-y1)=k(x-x1)变形，x=(y-y1)/k+x1

        else
            for(i=startY+1;i>=endY;i--)
            {
                r_border[i]=(int)((i - startY) / k + startX); //(y-y1)=k(x-x1)变形，x=(y-y1)/k+x1

                if(r_border[i] > right_stop_point[i])           //右截止线
                    r_border[i]=right_stop_point[i];

                else if(r_border[i] < left_stop_point[i])       //左截止线
                    r_border[i]=left_stop_point[i];
            }
    }

    else if (dir==-1)     //左线 L
    {
        startX = l_border[startY];

        if (k==0)       //斜率为0则是斜率为无穷大的情况，所以垂直向上拉线
            for(i=startY+1;i>=endY;i--)
                l_border[i]=startX; //(y-y1)=k(x-x1)变形，x=(y-y1)/k+x1

        else
            for(i=startY+1;i>=endY;i--)
            {
                l_border[i]=(int)((i - startY) / k + startX); //(y-y1)=k(x-x1)变形，x=(y-y1)/k+x1

                if(l_border[i] > right_stop_point[i])           //右截止线
                    l_border[i]=right_stop_point[i];

                else if(l_border[i] < left_stop_point[i])       //左截止线
                    l_border[i]=left_stop_point[i];

            }
    }
}



/*-------------------------------------------------------------------------------------------------------------------
  @brief     左补线
  @param     补线的起点，终点
  @return    null
  Sample     Left_Add_Line(int x1,int y1,int x2,int y2);
  @note      补的直接是边界，点最好是可信度高的,不要乱补
-------------------------------------------------------------------------------------------------------------------*/
void Left_Add_Line(int x1,int y1,int x2,int y2)//左补线,补的是边界
{
    int i;
    int hx;

    if(x1>=image_w-1)//起始点位置校正，排除数组越界的可能
        x1=image_w-1;
    else if(x1<=0)
        x1=0;

    if(y1>=image_h-1)
        y1=image_h-1;
    else if(y1<=0)
        y1=0;

    if(x2>=image_w-1)
        x2=image_w-1;
    else if(x2<=0)
        x2=0;
    if(y2>=MT9V03X_H-1)
        y2=MT9V03X_H-1;
    else if(y2<=0)
        y2=0;





    if (y1 > y2)   //若1点在下 （y值较大）
    {
        for (i=y1;i>=y2;i--)//根据斜率补线即可
        {
            hx=(i-y1)*(x2-x1)/(y2-y1)+x1;

            if(hx > right_stop_point[i])
                l_border[i]=right_stop_point[i];

            else if(hx < left_stop_point[i])
                l_border[i]=left_stop_point[i];

            else
                l_border[i]=hx;
        }
    }

    else   //若1点在上 （y值较小）
    {
        for (i=y1;i<=y2;i++)//根据斜率补线即可
        {
            hx=(i-y1)*(x2-x1)/(y2-y1)+x1;

            if(hx > right_stop_point[i])
                l_border[i]=right_stop_point[i];

            else if(hx < left_stop_point[i])
                l_border[i]=left_stop_point[i];

            else
                l_border[i]=hx;
        }
    }
}

/*-------------------------------------------------------------------------------------------------------------------
  @brief     右边界延长
  @param     延长起始行数，延长到某行
  @return    null
  Sample     Lengthen_Right_Boundry(int start,int end)；
  @note      从起始点向上找3个点，算出斜率，向下延长，直至结束点
-------------------------------------------------------------------------------------------------------------------*/
void Lengthen_Right_Boundry(int start, int end_line)
{
    int i,t;
    float k=0;

    if(start>image_h-1)//起始点位置校正，排除数组越界的可能
        start=image_h-1;
    else if(start<0)
        start=0;

    if(end_line>image_h-1)
        end_line=image_h-1;
    else if(end_line<=0)
        end_line=0;

    if (start >= end_line) return;  //当起点比终末行低，不工作

    if( start < 5 )//因为需要在开始点向上找3个点，对于起始点过于靠上，不能做延长，只能直接连线    && start < end
        Right_Add_Line(r_border[start], start, r_border[end_line], end_line);

    else
    {
        k = (float) (r_border[start]-r_border[start-3]) / 5.0;  //这里的k是1/斜率

        for(i=start;i<=end_line;i++)
        {
            r_border[i]=(int)(i-start)*k+r_border[start];   //(x=(y-y1)*k+x1),点斜式变形

            if(r_border[i] > right_stop_point[i])
                r_border[i] = right_stop_point[i];

            else if(r_border[i] < left_stop_point[i])
                r_border[i] = left_stop_point[i];
        }

    }
}

void huandao_control()
{
//    if (circle_direction_angle != -999)
//    {
//        yaw_angle_error=degree_y-circle_direction_angle;
//        if (yaw_angle_error>0 && circle_dir == 1) yaw_angle_error-=360;

//    }
//    else yaw_angle_error=0;

    if (circle_direction_angle != -999)
    {
        yaw_angle_error+=my_abs(degree_y-circle_direction_angle)>100?0:degree_y-circle_direction_angle;//
        circle_direction_angle=degree_y;
    }
    else yaw_angle_error=0;


    switch (yuan_flag)
    {
        case 1:             //进环(直道


            if (circle_dir == -1)       //左环
            {
                circle_in_distance-=real_speed_mm*0.001;

                if (circle_in_distance < 0) //走到了环中间
                {
                    yuan_flag = 2;
//                    circle_in_distance = 200;
                    offset_flag = -1;//循左线
                }
            }
            else if (circle_dir == 1)       //右环
            {
                circle_in_distance-=real_speed_mm*0.001;

                if (circle_in_distance < 0) //走到了环中间
                {
                    yuan_flag = 2;
//                    circle_in_distance = 200;
                    offset_flag = 1;//循右线
                }
            }

            break;



        case 2:     //进环(弯道
            if (circle_dir == -1)       //左环
            {
                if (yaw_angle_error>45 && yaw_angle_error<180)
                {
                    circle_in_distance=0;
                    yuan_flag = 3;
                    offset_flag = 0;
                    offset_flag = 1;
                }
            }

            else  if (circle_dir == 1)                      //右环
            {
                if (yaw_angle_error<-45 && yaw_angle_error>-180)
                {
                    circle_in_distance=0;
                    yuan_flag = 3;
                    offset_flag = 0;
                    offset_flag = -1;
                }
            }

            break;



        case 3:     //环内
            if (circle_dir == -1)       //左环
            {
                if (yaw_angle_error>220)
                {
                    offset_flag = -1;
                    yuan_flag = 4;
                }
            }
            else if (circle_dir == 1)   //右环
            {
                if (yaw_angle_error<-220)
                {
                    offset_flag = 1;
                    yuan_flag = 4;
                }
            }

            break;



        case 4:     //出环，走内环
            if (circle_dir == -1)       //左环
            {
                if (yaw_angle_error>320)
                {
                    offset_flag = 1;
                    circle_dir = 0;
//                    circle_bottom = 0;
                    yuan_flag = 5;
                }
            }
            else if (circle_dir == 1)   //右环
            {
                if (yaw_angle_error<-320)
                {
                    offset_flag = -1;
                    circle_dir = 0;
//                    circle_bottom = 0;
                    yuan_flag = 5;
                }
            }

            break;



        case 5:
            circle_in_distance+=real_speed_mm*0.001;
            if (circle_in_distance>yuan_out)
            {
                circle_in_distance=0;
                offset_flag = 0;
                yuan_flag = 6;
                circle_direction_angle = -999;
                acc_limit=acc_limit_common;
                dec_limit=dec_limit_common;
            }
            break;

        case 6:
            circle_in_distance+=real_speed_mm*0.001;
            if (circle_in_distance>yuan_cooldown)
            {
                circle_in_distance=0;
                yuan_flag = 0;
            }
            break;
    }
}
